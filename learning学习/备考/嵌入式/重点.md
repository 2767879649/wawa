1. 杂项设备结构体是<font color="#c00000">struct miscdevice</font>，注册函数是<font color="#c00000">miscregister</font>，注销函数是<font color="#c00000">miscderegister</font>。
2. input子系统由<font color="#c00000">驱动层</font>、<font color="#c00000">核心层</font>和<font color="#c00000">事件处理层</font>三部分组成。
3. 注册杂项设备时，需要填充miscdevice结构体的 <font color="#c00000">minor</font>、<font color="#c00000">name</font>和 <font color="#c00000">fops</font>三个成员。
	/* 定义杂项设备结构体，动态次设备号 */ 
	static struct miscdevice my_miscdev = { 
	.<font color="#92d050">minor</font> = MISC_DYNAMIC_MINOR, 
	.<font color="#92d050">n</font><font color="#92d050">ame</font> = "my_misc_dev", 
	.<font color="#92d050">fops</font> = &misc_fops, 
	};
4. <font color="#c00000">platform_device</font> 硬件设备 <font color="#c00000">miscdevice</font> 杂项设备
5. include<linux/module.h>
6. <font color="#c00000">MODULE_LICENSE("GPL");</font>声明必须写
7. 程序：
	1. #include <linux/module.h>
	2. #include <linux/platform_device.h>
	3. #include <linux/miscdevice.h>
	4. #include <linux/fs.h>
	5. //头文件
	6. 
	7. /* 空的 file_operations，仅用于注册，它只用来占位置、保证驱动合法，不实现任何读写功能。 */
	8. <font color="#c00000">static const struct file_operations misc_fops = {</font>
	9.    <font color="#c00000"> .owner = THIS_MODULE,</font>
	10. <font color="#c00000">};</font>
	11. 
	12. /* 定义杂项设备结构体，动态次设备号 */
	13. <font color="#c00000">static struct miscdevice my_miscdev = {</font>
	14.    <font color="#c00000"> .minor = MISC_DYNAMIC_MINOR,</font>
	15.    <font color="#c00000"> .name  = "my_misc_dev",</font>
	16.    <font color="#c00000"> .fops  = &misc_fops,</font>
	17. <font color="#c00000">};</font>
	18. //模块加载
	19. <font color="#c00000">static int __init my_misc_init(void)</font>
	20.<font color="#c00000"> {</font>

	25. <font color="#c00000">}</font>
	26. 
	27.//模块卸载
	27. <font color="#c00000">static void __exit my_misc_exit(void)</font>
	28. <font color="#c00000">{</font>
	29.     <font color="#c00000">misc_deregister(&my_miscdev);</font>
	30. // 打印函数名 + 行号
	31. <font color="#c00000">printk(KERN_INFO "%s:%d\n", __func__, __LINE__);</font>
	32. <font color="#c00000">}</font>
	33. //使用声明的函数//
	34. <font color="#c00000">module_init(my_misc_init);</font>
