#include <linux/module.h>
#include <linux/init.h>
#include <linux/input.h>
//头文件
static struct input_dev *key_input_dev;//定义全局 input 
static int _init input_init (void)
{
//分配
key_input_dev = <font color="#c00000">input_allocate_device()</font>;//标准分配函数
// 设置事件能力
set_bit(EV_KEY, key_input_dev->evbit);
set_bit(KEY_A, key_input_dev->keybit);
// 注册
input_register_device(key_input_dev);
}
//案件上报骨架函数
static void key_report_event(int press)
{
    input_report_key(key_input_dev, KEY_A, press);
    input_sync(key_input_dev);
}
static void _exit input_exit(void)
{
//注销与释放
input_unregister_device(key_input_dev); // 先注销
input_free_device(key_input_dev);       // 再释放内存
}

void signal_handler(int sig)
{
    if (sig == SIGINT) {
        if (fd >= 0) {
            write(fd, "0", 1);
            close(fd);
        }
        exit(0);
    }
}
