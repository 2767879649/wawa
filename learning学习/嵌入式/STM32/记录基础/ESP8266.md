# AT使用命令
AT+RST  //重启
## STA 模式
//ESP 作为客户端，连接路由器
AT
ATE0
AT+CWMODE=1
AT+CWJAP="CU_3CMU","nw89rrg2"
//等待打印 WIFI GOT IP
AT+CIPMUX=1
AT+CIPSERVER=1,8080
AT+CIFSR
## AP 模式
//ESP 作为服务端，手机连接 ESP
AT
ATE0
AT+CWMODE=2
AT+CWSAP="ESP8266","12345678",1,4 //开启 AP 热点
AT+CIPMUX=1  //开启连接
AT+CIPSERVER=1,8080
