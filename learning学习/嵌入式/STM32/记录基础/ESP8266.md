int ESP_Send(char*cmd,char*ack,int16_t time)
{
    for(int i=0;i<=5;i++)
    {
        while(Serial_RxFlag!=1&&i==5)
            {
                if((time--)<0)
                return 1;
            }
        Serial_SendString(cmd);
        Delay_ms(500);
        if(strcmp(wenbenData,ack)==0)
            break;
        
    }
    OLED_ShowString(4,1,"                  ");
    OLED_ShowString(4,1,wenbenData);
    Serial_RxFlag=0;
    return 0;
}
void ESP_Init()        //初始化函数
{
    Serial_Init();
    Delay_ms(2000);
    if(ESP_Send("AT\r\n","OK",1000)==1)
        ESP_flag=1;
    if(ESP_Send("ATE0\r\n","OK",1000)==1)
    ESP_flag=2;
    if(ESP_Send("AT+CWMODE=1\r\n","OK",1000)==1)
    ESP_flag=3;
    if(ESP_Send("AT+CWJAP=\"CU_3CMU\",\"nw89rrg2\"\r\n","WIFI GOT IP",10000)==1)
    ESP_flag=4;
    if(ESP_Send("AT+CIPMUX=1\r\n","OK",1000)==1)
    ESP_flag=5;
    if(ESP_Send("AT+CIPSERVER=1,8080\r\n","CONNECT",5000)==1)
    ESP_flag=6;
    if(ESP_Send("AT+CIFSR\r\n","CONNECT",5000)==1)
    ESP_flag=7;
    wenbenDataclear();
}
