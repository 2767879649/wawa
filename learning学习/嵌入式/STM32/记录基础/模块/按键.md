# 按键
*设置按键值来判断按键是否按下*
使用上拉输入模式 [[GPIO端口]]
```c
#include "stm32f10x.h"                  // Device header
#include "Delay.h"
#include "Led.h"
#define Key_GPIO GPIO_Pin_10    //更改按键引脚
void Key_Init()
{
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA,ENABLE);
    GPIO_InitTypeDef GPIO_Struct;
    GPIO_Struct.GPIO_Mode=GPIO_Mode_IPU;
    GPIO_Struct.GPIO_Pin=Key_GPIO;
    GPIO_Struct.GPIO_Speed=GPIO_Speed_50MHz;
    GPIO_Init(GPIOA,&GPIO_Struct);
    
}
int Key_Get(void)
{
static int last_state = 1; // 上一次状态（默认高电平）
static int debounce_cnt = 0;//按下后，计数消抖
static int count = 0;//松开后，计数消抖
static int key_pressed = 0; // 是否已处理本次按下
static int mode = 0;// 当前模式（0~2
int current = digitalRead(button);
// 检测下降沿（按下）
if (last_state == 1 && current == 0)
{
// 开始消抖计数
 debounce_cnt = 2; // 20ms 后确认（假设每 10ms 调用一次）
}
// 消抖计数递减
if (debounce_cnt > 0)
 {
 debounce_cnt--;
 if (debounce_cnt == 0 && current == 0)
 {
// 确认是有效按下，且未被处理
 if (!key_pressed)
 {
key_pressed = 1;
 mode++;
if (mode > 1)
 mode = 0;
 }
}
 }
// 检测释放（上升沿）
 if (last_state == 0 && current == 1)
 {
  count=200;
 }
 if(count>0)
 {
  count--;
  if (count == 0 && current == 1)
  {
    key_pressed = 0; // 允许下一次按下
  }
 }
 last_state = current;
 return mode;
}
```
