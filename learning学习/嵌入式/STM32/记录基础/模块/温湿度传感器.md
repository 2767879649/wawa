# 温湿度传感器
![[Pasted image 20260806135623.png|523]]
- 数据采集时需要有时间间隔，不加延迟函数，数据显示会卡死；
- 主机与温湿度传感器共用一条总线
```c
#include "stm32f10x.h"                  // Device header
#include "Delay.h"
// DHT11引脚定义
#define DHT_PIN     GPIO_Pin_0
#define DHT_PORT    GPIOA
#define RCC_DHT     RCC_APB2Periph_GPIOA
typedef struct
{
    uint8_t temp_int;   // 温度整数
    uint8_t temp_dec;   // 温度小数(DHT11固定0)
    uint8_t humi_int;   // 湿度整数
    uint8_t humi_dec;   // 湿度小数(DHT11固定0)
} DHT_DataDef;
// 输出模式
void DHT_SetOut(void)
{
    GPIO_InitTypeDef GPIO_strut;
    GPIO_strut.GPIO_Pin = DHT_PIN;
    GPIO_strut.GPIO_Mode = GPIO_Mode_Out_PP;
    GPIO_strut.GPIO_Speed = GPIO_Speed_50MHz;
    GPIO_Init(DHT_PORT, &GPIO_strut);
}
// 输入模式
void DHT_SetIn(void)
{
    GPIO_InitTypeDef GPIO_strut;
    GPIO_strut.GPIO_Pin = DHT_PIN;
    GPIO_strut.GPIO_Mode = GPIO_Mode_IPU;
    GPIO_Init(DHT_PORT, &GPIO_strut);
}
void DHT_Init(void)
{
    RCC_APB2PeriphClockCmd(RCC_DHT, ENABLE);
    DHT_SetOut();
    GPIO_SetBits(DHT_PORT, DHT_PIN); // 空闲高电平
}
// 主机启动信号：拉低18ms，释放20~40us
void DHT_SendStart(void)
{
    DHT_SetOut();
    GPIO_ResetBits(DHT_PORT, DHT_PIN);//主机拉低电平
    Delay_ms(18);
    GPIO_SetBits(DHT_PORT, DHT_PIN);//主机拉高电平
    Delay_us(30);
    DHT_SetIn();
}//开启信号
// 读取1bit
uint8_t DHT_ReadBit(void)
{
    // 等待低电平结束（每个bit起始低电平约50us）
    while(GPIO_ReadInputDataBit(DHT_PORT, DHT_PIN) == 0);  // 这里也会卡死
    // 测量高电平宽度
    uint32_t cnt = 0;
    while(GPIO_ReadInputDataBit(DHT_PORT, DHT_PIN) == 1) {
        Delay_us(1);
        if (++cnt > 100) break;   // 高电平最长约70us，超100us认为错误
    }
    return (cnt > 30) ? 1 : 0;   // 大于30us为1
}
// 读1字节
uint8_t DHT_ReadByte(void)
{
    uint8_t dat = 0;
    for(uint8_t i=0; i<8; i++)
    {
        dat <<= 1;
        dat |= DHT_ReadBit();
    }
    return dat;
}
// 读取完整数据，返回0正常
uint8_t DHT_ReadData(DHT_DataDef *dat)
{
    uint8_t buf[5];
    DHT_SendStart();
 while(GPIO_ReadInputDataBit(DHT_PORT, DHT_PIN)==1);
    while(GPIO_ReadInputDataBit(DHT_PORT, DHT_PIN)==0);
        while(GPIO_ReadInputDataBit(DHT_PORT, DHT_PIN)==1);
    for(int8_t i=0;i<5;i++)
    {
        buf[i] = DHT_ReadByte();
    }
    Delay_ms(20);

    // 读5字节：湿度整数、湿度小数、温度整数、温度小数、校验和
    // 校验和判断
    if(buf[0]+buf[1]+buf[2]+buf[3] == buf[4])
    {
        dat->humi_int = buf[0];
        dat->humi_dec = buf[1];
        dat->temp_int = buf[2];
        dat->temp_dec = buf[3];
        return 0;
    }
    return 1; // 校验失败
}
```
