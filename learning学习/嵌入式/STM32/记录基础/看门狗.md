## 独立看门狗
时钟不需要手动开启
```c
int main()
{
    OLED_Init();
    if(RCC_GetFlagStatus(RCC_FLAG_IWDGRST)==SET)
    {
        OLED_ShowString (2,1,"123124");
        Delay_ms(500);
        OLED_ShowString (2,1,"       ");
        Delay_ms(100);
        RCC_ClearFlag();
    }
    IWDG_WriteAccessCmd(IWDG_WriteAccess_Enable);  //开启看门狗写使能
    IWDG_SetPrescaler(IWDG_Prescaler_16);//分频
    IWDG_SetReload(2499);//设置计数置，喂狗超时时间
    IWDG_ReloadCounter();    //喂狗
    IWDG_Enable();        //看门狗使能
    while(1)
    {    WWDG_SetCounter(54|0x40);
        Key_GetNum();
        OLED_ShowString (4,1,"asdfd");
        Delay_ms(5000);
        OLED_ShowString (4,1,"       ");
        Delay_ms(100);
     }
}

```

## 窗口看门狗
时钟需要手动开启
```c
int main()
{
    OLED_Init();
    if(RCC_GetFlagStatus(RCC_FLAG_WWDGRST)==SET)
    {
        OLED_ShowString (2,1,"123124");
        Delay_ms(500);
        OLED_ShowString (2,1,"       ");
        Delay_ms(100);
        RCC_ClearFlag();
    }
RCC_APB1PeriphClockCmd(RCC_APB1Periph_WWDG,ENABLE); //开启窗口看门狗时钟
WWDG_SetPrescaler(WWDG_Prescaler_8); //设置分频
WWDG_SetWindowValue(0x40|21);//最早时间
WWDG_Enable(0x40|54);        //开启看门狗，超时时间
        while(1)
    {    WWDG_SetCounter(54|0x40);
        Key_GetNum();
        OLED_ShowString (4,1,"asdfd");
        Delay_ms(5000);
        OLED_ShowString (4,1,"       ");
        Delay_ms(100);
     }
}
```
