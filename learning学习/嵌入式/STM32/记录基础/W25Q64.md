# **W25Q64简介**

- W25Qxx系列是一种低成本、小型化、使用简单的非易失性存储器，常应用于数据存储、字库存储、固件程序存储等场景
- 存储介质：Nor Flash（闪存）
- 时钟频率：80MHz / 160MHz (Dual SPI) / 320MHz (Quad SPI)

# **Flash操作注意事项**
- **写入操作时：**
- - 写入操作前，必须<font color="#9bbb59">先进行写使能</font>(写入操作完成后会自动关闭)
- - 每个数据位只能由1改写为0，不能由0改写为1
- - 写入<font color="#9bbb59">数据前必须先擦除</font>，擦除后，所有数据位变为1
- - 擦除必须按最小擦除单元进行
- - 连续写入多字节时，最多写入一页的数据，超过页尾位置的数据，会回到页首覆盖写入
- - 写入操作结束后，芯片进入忙状态，不响应新的读写操作
- **读取操作时：**
- - 直接调用读取时序，无需使能，无需额外操作，没有页的限制，读取操作结束后不会进入忙状态，但不能在忙状态时读取


<font color="#9bbb59">数据不能跨页写入，到达页尾</font>
<font color="#9bbb59">地址后会返回页首地址，可以跨页读出</font>


# 软件初始化
[[SPI通信]] <----SPI软件初始化
```c
void W25Q64_Init()
{
    MySPI_Init();                                    //初始化
}
void W25Q64_ReadID(uint8_t *MID,uint16_t *DID)
{
    MySPI_Start();                                //开始位
    MySPI_SwapByte(W25Q64_JEDEC_ID);            //发送读取指令集
    *MID=MySPI_SwapByte(W25Q64_DUMMY_BYTE);       //交换地址
    *DID=MySPI_SwapByte(W25Q64_DUMMY_BYTE);        //交换高8位信息
    *DID<<=8;                                    //移位
    *DID |=MySPI_SwapByte(W25Q64_DUMMY_BYTE);    //交换低8位信息，并或上高8位
    MySPI_Stop();                                   //停止位
}//获取设备ID
void W25Q64_WeiteEnable(void)
{
    MySPI_Start();                                
    MySPI_SwapByte(W25Q64_WRITE_ENABLE);           //发送写开启使能的指令
    MySPI_Stop();
}//开启写使能，每次读或写时均需要开启写使能
void W25Q64_WaitBusy(void)
{
    MySPI_Start();
    MySPI_SwapByte(W25Q64_READ_STATUS_REGISTER_1);    
    while((MySPI_SwapByte(0xFF)&0x01)==0x01);    //最低位为忙状态位，为1说明1忙
    MySPI_Stop();
}//等待CPU忙

void W25Q64_PageProgram(uint32_t Address,uint16_t *Data,uint16_t Count)
{
    W25Q64_WeiteEnable();        //写使能，每次完成写入后会自动关闭
    W25Q64_WaitBusy();            //忙等待
    uint16_t i;
    MySPI_Start();
    MySPI_SwapByte(W25Q64_PAGE_PROGRAM);        //页编程指令
    MySPI_SwapByte(Address>>16);            
    MySPI_SwapByte(Address>>8);
    MySPI_SwapByte(Address);            //发送地址，一次因为一次只发送8位字节，就是两位数据，所以需要移位，先发高位
    for(i=0;i<Count;i++)
    {
    MySPI_SwapByte(Data[i]);             //写入数据
    }
    MySPI_Stop();
}//页编程，写入数据
void W25Q64_SectorErase(uint32_t Address)
{
    W25Q64_WeiteEnable();            //写使能
    MySPI_Start();
    MySPI_SwapByte(W25Q64_SECTOR_ERASE_4KB);        //擦除指令
    MySPI_SwapByte(Address>>16);
    MySPI_SwapByte(Address>>8);
    MySPI_SwapByte(Address);                //地址
    MySPI_Stop();
    W25Q64_WaitBusy();
}//块擦除
void W25Q64_ReadData(uint32_t Address,uint16_t *Data,uint32_t Count)
{
    W25Q64_WaitBusy();
    uint32_t i;
    MySPI_Start();
    MySPI_SwapByte(W25Q64_READ_DATA);        //读取指令
    MySPI_SwapByte(Address>>16);
    MySPI_SwapByte(Address>>8);
    MySPI_SwapByte(Address);                //地址
    for(i=0;i<Count;i++)
    {
    Data[i]=MySPI_SwapByte(0xFF);         //读取数据
    }
    MySPI_Stop();
}//读取数据

```
