# USART
**USART简介**
- USART（Universal Synchronous/Asynchronous Receiver/Transmitter）通用同步/异步收发器
- USART是STM32内部集成的硬件外设，可根据数据寄存器的一个字节数据自动生成数据帧时序，从TX引脚发送出去，也可自动接收RX引脚的数据帧时序，拼接为一个字节数据，存放在数据寄存器里
- 自带波特率发生器，最高达4.5Mbits/s
- 可配置数据位长度（8/9）、停止位长度（0.5/1/1.5/2）
- 可选校验位（无校验/奇校验/偶校验）
- 支持同步模式、硬件流控制、DMA、智能卡、IrDA、LIN
- STM32F103C8T6 USART资源：<font color="#245bdb">USART1、USART2、USART3</font>
**电平标准**
电平标准是数据1和数据0的表达方式，是传输线缆中人为规定的电压与数据的对应关系。串口常用的电平标准有如下三种：
- **TTL电平**(常用)：+3.3V或+5V表示1，0V表示0
- **RS232电平**：-3～-15V表示1，+3～+15V表示0
- **RS485电平**：两线压差+2～+6V表示1，-2～-6V表示0（差分信号）

## printf 使用
在魔术棒编辑器中添加 `--no-multibyte-chars` 就可以串口发送中文
1. printf 重定向
```c
int fputc(int ch,FILE *f)
{
    Serial_SendByte(ch);
    return ch;
}//main.c中直接printf()使用，不需要声明
```
2. 发送字符串
```c
void Serial_Printf(char *format, ...)
{
    char String[100];                //定义字符数组
    va_list arg;                    //定义可变参数列表数据类型的变量arg
    va_start(arg, format);            //从format开始，接收参数列表到arg变量
    vsprintf(String, format, arg);    //使用vsprintf打印格式化字符串和参数列表到字符数组中
    va_end(arg);                    //结束变量arg
    Serial_SendString(String);        //串口发送字符数组（字符串）
}
//声明后，在main.c中使用函数Serial_Printf("你好，世界",123);

```
## 初始化+开启中断
```c
void Serial_Init(void)
{
        
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_USART1,ENABLE);
    RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA,ENABLE);
    GPIO_InitTypeDef GPIO_Struct;
    GPIO_Struct.GPIO_Mode=GPIO_Mode_AF_PP;
    GPIO_Struct.GPIO_Pin=GPIO_Pin_9;
    GPIO_Struct.GPIO_Speed=GPIO_Speed_50MHz;
    GPIO_Init(GPIOA,&GPIO_Struct);                //GPIO初始化
    
    USART_InitTypeDef USART_Struct;            //结构体
    USART_Struct.USART_BaudRate=9600;            //设置波特率
    USART_Struct.USART_HardwareFlowControl=USART_HardwareFlowControl_None;         //硬件流控，不需要
    USART_Struct.USART_Mode=USART_Mode_Tx|USART_Mode_Rx;            //设置为发送接收模式
    USART_Struct.USART_Parity=USART_Parity_No;                 //奇偶校验位，不需要
    USART_Struct.USART_StopBits=USART_StopBits_1;             //停止位占多少
    USART_Struct.USART_WordLength=USART_WordLength_8b;         //数据位占多少
    USART_Init(USART1,&USART_Struct);                    //USART初始化

    USART_ITConfig(USART1,USART_IT_RXNE,ENABLE);        //开启USART的中断
    NVIC_PriorityGroupConfig(NVIC_PriorityGroup_2);    //NVIC分组
    NVIC_InitTypeDef NVIC_Strurt;                    //NVIC结构体
    NVIC_Strurt.NVIC_IRQChannel=USART1_IRQn;        //NVIC的中断线路
    NVIC_Strurt.NVIC_IRQChannelCmd=ENABLE;            //NVIC线路使能
    NVIC_Strurt.NVIC_IRQChannelPreemptionPriority=1;        //设置抢占式优先级
    NVIC_Strurt.NVIC_IRQChannelSubPriority=1;        //设置响应式优先级
    NVIC_Init(&NVIC_Strurt);                    //初始化
    
    USART_Cmd(USART1,ENABLE);                                //开启USART串口
}
```
### 串口数据函数
*发送字节*
```c
void Serial_SendByte(uint8_t Byte)
{
    USART_SendData(USART1,Byte);
    while(USART_GetFlagStatus(USART1,USART_FLAG_TXE)==RESET);
}

```
发送数组
```c
void Serial_SendArray(uint8_t *Array,uint16_t Length)
{
    uint16_t i;
    for(i=0;i<Length;i++)
    {
        Serial_SendByte(Array[i]);
    }
}
```
*发送字符串*
```c
void Serial_SendString(char *String)
{
    uint8_t i;
    for(i=0;String[i]!='\0';i++)
    {
        Serial_SendByte(String[i]);
    }
    
}
```
*发送一串数字*
```c
uint32_t Serial_Pow(uint32_t X,uint32_t Y)
{
    uint32_t Result =1;
    while(Y--)
    {
        Result*=X;
    }
    return Result;
}
void Serial_SendNumber(uint32_t Number,uint8_t Length)
{
    uint8_t i;
    for(i=0;i<Length;i++)
    {
        Serial_SendByte(Number/Serial_Pow(10,Length-i-1)%10+'0');
    }
}
```
