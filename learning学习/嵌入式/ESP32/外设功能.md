# LEDC 通道(PWM 硬件外设)
1. ledcSetup(通道，频率，分辨率)；//设置通道
2. ledcAttachPin(引脚，通道)；//将引脚和通道绑定
3. ledcWrite（通道，i）；更改 PWM 输出

# ADC
1. analogRead(引脚)                       //ADC 读取
2. analogReadResolution(12);                     // 设置ADC分辨率
3.  analogSetAttenuation(ADC_11db);         // 设置ADC输入衰减

# 串口
1. Serial.begin(115200);           // 设置波特率
2.   Serial.printf("输出值:%d\n", ADC);        //串口打印
3. Serial.print(ADC)；             //打印数字

# 外部中断
1. pinMode(GPIO_4, INPUT_PULLUP);                                              // 设置引脚
2.   attachInterrupt(digitalPinToInterrupt(GPIO_4), interrupts_handle, FALLING); // 设置中断引脚

# 定时器
### 硬件定时器触发
1. `hw_timer_t *timer = NULL;` // 创建定时器指针
2. `timer = timerBegin(0, 80, true);` //初始化定时器
3. `timerAttachInterrupt(timer, timer_interrupt, true);` //将定时器指针与定时器中断函数绑定
4. `timerAlarmWrite(timer, 1000 * 1000, true)`//设置定时器模式
5. `timerAlarmEnable(timer)`//启动定时器
