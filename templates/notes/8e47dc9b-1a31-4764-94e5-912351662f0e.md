#include <linux/module.h>
#include <linux/platform_device.h>
#include <linux/miscdevice.h>
#include <linux/fs.h>
//头文件

/* 空的 file_operations，仅用于注册，它只用来占位置、保证驱动合法，不实现任何读写功能。 */
static const struct file_operations misc_fops = {
    .owner = THIS_MODULE,
};

/* 定义杂项设备结构体，动态次设备号 */
static struct miscdevice my_miscdev = {
    .minor = MISC_DYNAMIC_MINOR,
    .name  = "my_misc_dev",
    .fops  = &misc_fops,
};
//模块加载
static int __init my_misc_init(void)
{
    int ret;
    // 打印函数名 + 行号
    printk(KERN_INFO "%s:%d: misc device init\n", __func__, __LINE__);
    ret = misc_register(&my_miscdev);
    return 0;
}

//模块卸载
static void __exit my_misc_exit(void)
{
    // 打印函数名 + 行号
    printk(KERN_INFO "%s:%d: misc device exit\n", __func__, __LINE__);
    misc_deregister(&my_miscdev);
}
//使用声明的函数//
module_init(my_misc_init);
module_exit(my_misc_exit);
//必写声明
MODULE_LICENSE("GPL");
