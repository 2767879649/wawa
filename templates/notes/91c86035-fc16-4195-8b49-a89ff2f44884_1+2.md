#include <ioCC2530.h>

unsigned int t1_cnt = 0;

void Delay_ms(unsigned int ms)
{
    unsigned int i, j;
    for(i = 0; i < ms; i++)
        for(j = 0; j < 535; j++);
}

void GPIO_Init(void)
{
    // LED在P1口
    P1SEL &= ~0x03;    // P1_0/P1_1 通用IO
    P1DIR |= 0x03;     // P1_0(LED1)、P1_1(LED2) 输出
    P1_0 = 1;          // LED1默认灭
    P1_1 = 1;          // LED2默认灭

    // S1在P0_1（按实际改）
    P0SEL &= ~0x02;    // P0_1 通用IO
    P0DIR &= ~0x02;    // P0_1 输入
    P0INP &= ~0x02;    // P0_1 打开上下拉
    P2INP &= ~0x20;    // P0口上拉(bit5=PDUP0=0)，按键空闲高电平
}

void Timer1_Init(void)
{
    T1CTL = 0x0D;      // 128分频 + 自由运行
    IRCON = 0;
}

void Key_Int_Init(void)
{
    PICTL |= 0x01;     // P0_0~P0_7 下降沿触发(bit0=P0ICON)
    P0IEN |= 0x02;     // 使能 P0_1 引脚中断
    IEN1  |= 0x20;     // 使能 P0 端口中断(P0IE, IEN1.bit5)
    P0IFG  = 0x00;     // 清引脚标志
    EA = 1;            // 打开总中断
}

void main(void)
{
    GPIO_Init();
    Timer1_Init();
    Key_Int_Init();

    while(1)
    {
        // 功能1：查询定时器，LED1慢闪0.5Hz
        if(IRCON > 0)
        {
            IRCON = 0;
            t1_cnt++;
            if(t1_cnt >= 2)
            {
                t1_cnt = 0;
                P1_0 = ~P1_0;
            }
        }
    }
}

// 功能2：P0端口中断服务函数
#pragma vector = P0INT_VECTOR
__interrupt void P0_ISR(void)
{
    unsigned char i;

    P0IEN &= ~0x02;       // 先关S1中断，防抖动反复触发

    if(P0IFG & 0x02)      // 确认P0_1(S1)触发
    {
        P1_1 = ~P1_1;     // LED2翻转

        for(i = 0; i < 3; i++)   // LED1快闪3次
        {
            P1_0 = 0;
            Delay_ms(100);
            P1_0 = 1;
            Delay_ms(100);
        }
    }

    P0IFG = 0;            // 清P0引脚标志
    P0IF  = 0;            // 清P0端口标志(IRCON.P0IF)
    P0IEN |= 0x02;        // 重新使能S1中断
}
