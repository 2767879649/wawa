static uint8_t recv[1024];
static uint8_t recv_index = 0;
static uint32_t starttime;
void ESP_Send(char *cmd, uint16_t Time)
{
  sprintf(recv, "%s\r\n", cmd);
  HAL_UART_Transmit(&huart1, (uint8_t *)recv, strlen(recv), Time);
}

void ESP_Init(void)
{
  static uint8_t flag = 0;
  while (1)
  {
    if (recv_index != flag)
    {
      flag = recv_index;
      memset(recv, 0, sizeof(recv));
    }
    if (flag == 0)
    {
      memset(recv, 0, sizeof(recv));
      ESP_Send("AT", 1000);
    }
    else if (flag == 1)
    {

      ESP_Send("AT+CWMODE=1", 1000);
    }
    else if (flag == 2)
    {
      starttime = HAL_GetTick();
      ESP_Send("AT+CWJAP=\"CU_3CMU\",\"nw89rrg2\"", 1000);
      while (1)
      {
        if (strstr((char *)recv, "\r\nOK") != NULL)
        {
          break;
        }
        else if (HAL_GetTick() - starttime > 5000)
        {
          break;
        }
      }
    }

    else if (flag == 3)
    {
      ESP_Send("AT+CIPMUX=0", 100);
      HAL_Delay(500);
    }
    else if (flag == 4)
    {
      ESP_Send("AT+CIPSTART=\"TCP\",\"api.seniverse.com\",80", 100);
      HAL_Delay(500);
    }
    else if (flag == 5)
    {
      ESP_Send("AT+CIPMODE=1", 100);
      HAL_Delay(500);
    }
    else if (flag == 6)
    {
      ESP_Send("AT+CIPSEND", 100);
      HAL_Delay(500);
    }
    else if (flag == 7)
    {
      ESP_Send("GET /v3/weather/now.json?key=SgCh-_e58GLicOBpJ&location=beijing&language=en", 500);
      HAL_Delay(1000);
      break;
    }
    OLED_ShowInt(6, 20, recv_index, 8);
    printf("%s\r\n", recv); // 或者用 HAL_UART_Transmit 发送
    HAL_Delay(1000);        // 给串口发送留点时间
    if (strstr((char *)recv, "\r\nOK") != NULL)
    {
      recv_index++;
    }
    memset(recv, 0, sizeof(recv));
    OLED_ShowString(6, 0, "      ", 8);
  }
}
